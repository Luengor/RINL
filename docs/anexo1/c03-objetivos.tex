\chapter{Objetivos del sistema}
\label{anexo1:cap:objetivos}
En este capítulo se definen los objetivos que se pretenden alcanzar con el desarrollo de este
trabajo de fin de grado.

\objetivo{
    id = {0001},
    nombre = {Detección de posturas},
    descripcion = {
        El sistema deberá ser capaz de realizar una detección y seguimiento de la postura de una
        persona en tiempo real utilizando una cámara.
    },
}{}

\objetivo{
    id = {0002},
    nombre = {Juegos y actividades},
    descripcion = {
        El sistema deberá ofrecer una seríe de juegos y actividades interactivas que utilicen la
        detección de posturas como método de interacción.
    },
}{
    subobjetivos = {
        % O el 1 depende del 2 o viceversa, no sé
    }
}

\objetivo{
    id = {0003},
    nombre = {Adaptación a diferentes usuarios},
    descripcion = {
        El sistema deberá ofrecer personalización para adaptarse a las capacidades físicas de cada
        usuario, adaptando el nivel de dificultad.
    }
}{}

\objetivo{
    id = {0004},
    nombre = {Interfaz WEB},
    descripcion = {
        El sistema deberá ofrecer una interfaz web que permita a los usuarios interactuar con el
        sistema, configurar sus preferencias y acceder a los juegos y actividades.    
    }
}{}

\objetivo{
    id = {0005},
    nombre = {Estadísticas y seguimiento},
    descripcion = {
        El sistema deberá ofrecer un sistema de seguimiento y estadísiticas que permita al usuario
        almacenar un historial de sus actividades y sesiones de juego.
    }
}{}

\objetivo{
    id = {0006},
    nombre = {Gestión del acceso},
    descripcion = {
        El sistema deberá ofrecer un sistema de gestión de acceso que permita a los usuarios
        iniciar y cerrar sessión, así como acceder a su información personal y estadísticas.
    }
}{}
