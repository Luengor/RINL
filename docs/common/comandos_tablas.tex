\newcommand{\bloque}[1]{
    \Block[l]{1-2}{#1}

\newcommand{\tableshell}[4]{
    \begin{table}[H]
    \protected@edef\@currentlabelname{#3}
    \label{#4}
    \begingroup
    \centering
    \NiceMatrixOptions{
        cell-space-limits = 3pt
    }
    \makebox[\textwidth] {
    \begin{NiceTabular}[width=15cm]{ #1 }
    \CodeBefore
        \rowcolor{Snow3}{1}
        \rowcolors{2}{Snow2}{Snow1}[respect-blocks]
    \Body
    \hline
        #2
    \hline
    \end{NiceTabular}
    }
    \caption{#3}
    \endgroup
    \end{table}
}

\newcounter{csec}
\newcounter{cexc}
\newcounter{cren}
\newcounter{ncasouso}
\setcounter{ncasouso}{1}

\newcommand{\casouso}[2]{
    \setkeys{keygenerica}{#1}
    \setkeys{casouso}{#2}

    \def\parwidth{11.5cm}
    \setcounter{csec}{\casouso@longsecuencia}
    \stepcounter{csec}

    \setcounter{cexc}{\casouso@longexcepciones}
    \stepcounter{cexc}

    \setcounter{cren}{\casouso@longrendimiento}
    \stepcounter{cren}

    \protected@edef\ncasouso{%
        \num[mode=text,reset-text-series=false,minimum-integer-digits=4]{\thencasouso}
    }
    
    \tableshell{ | p{2.75cm} | X[c,m] | X[10,m] | }{
        \textbf{UC-\ncasouso} & \bloque{\textbf{\keygenerica@nombre}} & \\ 
        \hline
        \textbf{Versión} & \bloque{1.0 (\textbf{TODO: FECHA})} \\
        \textbf{Autores} & \bloque{Luengo Román, Alberto} \\
        \textbf{Fuentes} & \bloque{\keygenerica@fuentes} \\
        \textbf{Objetivos asociados} & \bloque{\begin{minipage}{\parwidth}
            \begin{itemize}[left=0pt, itemsep=0pt, topsep=0pt]
                \casouso@objetivosAsociados
            \end{itemize}
        \end{minipage}} \\
        \textbf{Requisitos asociados} & \bloque{\begin{minipage}{\parwidth}
            \begin{itemize}[left=0pt, itemsep=0pt, topsep=0pt]
                \casouso@requisitosAsociados
            \end{itemize}
        \end{minipage}} \\
        \textbf{Descripción} & \bloque{\begin{minipage}{\parwidth}\keygenerica@descripcion\end{minipage}} \\ 
        \textbf{Precondición} & \bloque{\begin{minipage}{\parwidth}\casouso@precondicion\end{minipage}} \\

        \Block[l]{\thecsec-1}{\textbf{Secuencia normal}} & \textbf{Paso} & \textbf{Acción} \\
        \cline{2-3}
        \casouso@secuencia
        \textbf{Postcondición} & \bloque{\begin{minipage}{\parwidth}\casouso@postcondicion\end{minipage}} \\

        \ifx\casouso@excepciones\empty
        \else
            \Block[l]{\thecexc-1}{\textbf{Excepciones}} & \textbf{Paso} & \textbf{Acción} \\
            \cline{2-3}
            \casouso@excepciones
        \fi

        \ifx\casouso@rendimiento\empty
        \else
            \Block[l]{\theren-1}{\textbf{Rendimiento}} & \textbf{Medida} & \textbf{Valor} \\
            \cline{2-3}
            \casouso@rendimiento
        \fi
        \textbf{Frecuencia} & \bloque{\casouso@frecuencia} \\
        \textbf{Importancia} & \bloque{\keygenerica@importancia} \\
        \textbf{Urgencia} & \bloque{\keygenerica@urgencia} \\
        \textbf{Estado} & \bloque{\keygenerica@estado} \\
        \textbf{Estabilidad} & \bloque{\keygenerica@estabilidad} \\
        \textbf{Comentarios} & \bloque{\keygenerica@comentarios} \\
    }{%
        UC-\ncasouso \keygenerica@nombre
    }
    {anexo1:tab:cu:\casouso@key}

    \stepcounter{ncasouso}
}
