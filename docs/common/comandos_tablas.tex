\newcommand{\bloque}[1]{
    \Block[l]{1-2}{#1}
}

\define@key{tableshell}{cellSpaceLimits}{\def\tableshell@CellSpaceLimits{#1}}
\define@key{tableshell}{fontSize}{\def\tableshell@FontSize{#1}}
\define@key{tableshell}{width}{\def\tableshell@width{#1}}

\presetkeys{tableshell}{
    cellSpaceLimits = 4pt,
    fontSize = \normalsize,
    width = 15cm,
}{}

\NewDocumentCommand{\tableshell}{ O{ | l | X | } O{} m m m}{
    \setkeys{tableshell}{#2}

    \begin{table}[H]
    \protected@edef\@currentlabelname{#4}
    \label{#5}
    \begingroup
    \centering
    \NiceMatrixOptions{
        cell-space-limits = \tableshell@CellSpaceLimits,
    }
    \tableshell@FontSize
    \makebox[\textwidth] {
    \begin{NiceTabular}[width=\tableshell@width]{ #1 }
    \CodeBefore
        \rowcolor{Snow3}{1}
        \rowcolors{2}{Snow2}{Snow1}[respect-blocks]
    \Body
    \hline
        #3
    \hline
    \end{NiceTabular}
    }
    \caption{#4}
    \endgroup
    \end{table}
}
