\makeatletter

\define@key{diagrama}{nombre}{\def\diagrama@nombre{#1}}
\define@key{diagrama}{archivo}{\def\diagrama@archivo{#1}}
\define@key{diagrama}{etiqueta}{\def\diagrama@etiqueta{#1}}
\define@key{diagrama}{width}{\def\diagrama@width{#1}}
\define@key{diagrama}{forzarPosicion}{\def\diagrama@forzarPosicion{#1}}

\presetkeys{diagrama}{
    nombre = {Diagrama},
    archivo = {},
    etiqueta = {fig:diagrama},
    width = 0.8\textwidth,
    forzarPosicion = {}, 
}{}

\newcommand{\diagrama}[1]{
    \setkeys{diagrama}{#1}

    \ifx\diagrama@forzarPosicion\empty
        \begin{figure}[!ht]
    \else
        \begin{figure}[H]
    \fi
        \centering
        \includegraphics[width=\diagrama@width]{\diagrama@archivo}
        \caption{\diagrama@nombre}
        \label{\diagrama@etiqueta}
    \end{figure}
}

\makeatother
