\newglossaryentry{backend}
{

    },
    plural={backends},
}

{
    name={frontend},

    },
    plural={frontends},
}

\newglossaryentry{ORM}
{
    name={ORM},
    description={
        El Mapeo Relacional de Objetos (ORM) es una técinica de programación para convertir datos

        datos relacional.
    },
    type=\acronymtype,
    first={Mapeo Relacional de Objetos (ORM)},
    plural={Mapeos Relacionales de Objetos},
}

\newglossaryentry{dao}
{
    name={DAO},
    description={
        El patrón de diseño \glsfirst{dao} es un patrón que proporciona una interfaz para acceder
        a los datos de una aplicación, separando la lógica de negocio del acceso a los datos.
    },
    type=\acronymtype,
    first={Data Access Object (DAO)},
    plural={Data Access Objects (DAO)},
}

\newglossaryentry{api}
{
    name={API},
    description={
        Interfaz de Programación de Aplicaciones (API) es un conjunto de definiciones y protocolos
        que permiten la comunicación entre diferentes aplicaciones o componentes de software.
    },
    type=\acronymtype,
    first={Interfaz de Programación de Aplicaciones (API)},
    plural={APIs},
}

\newglossaryentry{hash}
{
    name={Hash},
    description={
        Un algoritmo de hash es una función que toma una entrada y devuelve una salida de longitud fija.
        Una función de hash debe ser unidireccional, es decir, no debe ser posible revertir el proceso
        para obtener la entrada original a partir de la salida.
    }
}

\newglossaryentry{xss}
{
    name={XSS},
    description={
        El Cross-Site Scripting (XSS) es una vulnerabilidad de seguridad que permite a un atacante
        inyectar scripts maliciosos en páginas web vistas por otros usuarios.
    },
    type=\acronymtype,
    first={Cross-Site Scripting (XSS)},
}

\newglossaryentry{csrf}
{
    name={CSRF},
    description={
        El Cross-Site Request Forgery (CSRF) es un tipo de ataque que permite a un atacante
        realizar acciones no autorizadas en nombre de un usuario autenticado en una aplicación web.
    },
    type=\acronymtype,
    first={Cross-Site Request Forgery (CSRF)},
}

\newglossaryentry{tsx}
{
    name={TSX},
    description={
        JSX es una extensión de sintaxis para JavaScript que permite escribir código HTML
        dentro de archivos JavaScript. TSX es la versión de JSX para TypeScript, que
        permite utilizar tipos estáticos y otras características de TypeScript en el código JSX.
    },
    type=\acronymtype
}

\newglossaryentry{asgi}
{
    name = {ASGI},
    description = {
        El Asynchronous Server Gateway Interface (ASGI) es un estándar para aplicaciones web

    },

}
