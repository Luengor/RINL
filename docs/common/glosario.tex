\newglossaryentry{backend}
{
    name={backend},
    description={
        Parte del software que se encarga de la lógica de negocio, el acceso a datos y la
        comunicación con el frontend.
    },
    plural={backends},
}

\newglossaryentry{frontend}
{
    name={frontend},
    description={
        Parte del software que se encarga de la interfaz de usuario y la interacción con el usuario.
    },
    plural={frontends},
}

\newglossaryentry{framework}
{
    name={framework},
    description={
        Conjunto de herramientas y convenciones que facilitan el desarrollo de software.
    },
    plural={frameworks},
}

\newglossaryentry{ORM}
{
    name={ORM},
    description={
        El Mapeo Relacional de Objetos (ORM) es una técinica de programación para convertir datos
        entre el sistema de tipos de un lenguaje de programación y los almacenados en una base de
        datos relacional.
    },
    type=\acronymtype,
    first={Mapeo Relacional de Objetos (ORM)},
    plural={Mapeos Relacionales de Objetos},
}

\newglossaryentry{dao}
{
    name={DAO},
    description={
        El patrón de diseño \glsfirst{dao} es un patrón que proporciona una interfaz para acceder
        a los datos de una aplicación, separando la lógica de negocio del acceso a los datos.
    },
    type=\acronymtype,
    first={Data Access Object (DAO)},
    plural={Data Access Objects (DAO)},
}

\newglossaryentry{api}
{
    name={API},
    description={
        Interfaz de Programación de Aplicaciones (API) es un conjunto de definiciones y protocolos
        que permiten la comunicación entre diferentes aplicaciones o componentes de software.
    },
    type=\acronymtype,
    first={Interfaz de Programación de Aplicaciones (API)},
    plural={API},
}

\newglossaryentry{hash}
{
    name={hash},
    description={
        Un algoritmo de hash es una función que toma una entrada y devuelve una salida de longitud fija.
        Una función de hash debe ser unidireccional, es decir, no debe ser posible revertir el proceso
        para obtener la entrada original a partir de la salida.
    }
}
