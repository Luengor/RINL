\section{Objetivo}
% Objeto, Requisitos iniciales, Hipótesis

% Justo después de venir de hablar de dispositivos de entrada y salida,
% empezamos a hablar de objetivos. El objetivo del proyecto es probar que la
% viabilidad de una plataforma de entretenimiento basada en la detección de
% posturas es posible.

\begin{frame}
    \frametitle{Objetivo principal}
    \begin{block}{Objetivo principal}
        Crear una \alert{plataforma de entretenimiento y ejercicio físico} que
        utilice la \alert{detección de posturas} del cuerpo humano como
        mecanismo principal de control.
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Objetivos secundarios}

    \begin{itemize}
        \item \alert{Adaptar} la plataforma a las capacidades físicas de cada
        usuario.
        \pause

        \item Hacer de dicha plataforma una \alert{aplicación web accesible}
        desde cualquier dispositivo con conexión a Internet y cámara web.
        \pause

        \item Proporcionar \alert{métricas y estadísticas} de rendimiento y
        forma física.
        \pause

        \item Garantizar la \alert{privacidad y seguridad} de los datos
        personales de los usuarios.

    \end{itemize}
\end{frame}
