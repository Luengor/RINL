% Como se ha hecho la planificación
\begin{frame}
    \frametitle{Planificación del proyecto}

    \centering
    \begin{columns}
        \column{0.6\textwidth}
        \begin{itemize}
            \item Se han realizado \textit{sprints} con \alert{duración de una
            semana}.

            \item Todos los roles del equipo han sido asumidos por el autor.

            \item No se han realizado reuniones diarias ni revisiones de
            \textit{sprint}.

            \item Se ha utilizado la estimación de tareas mediante \alert{puntos
            de historia} y la planificación de tareas mediante un tablero
            similar al de Kanban.

        \end{itemize}

        \column{0.4\textwidth}
        \includegraphics[height=0.45\textheight]{pre/imgs/meto/jira.png}
    \end{columns}
\end{frame}
